```
class Solution:
    def minOperations(self, s: str) -> int:
        # Check if already sorted
        if all(s[i] <= s[i + 1] for i in range(len(s) - 1)):
            return 0

        n = len(s)
        # Length 2 cannot be sorted by any proper substring
        if n == 2:
            return -1

        min_char = min(s)
        max_char = max(s)

        # 1 operation
        if s[0] == min_char or s[n - 1] == max_char:
            return 1

        # 2 operations vs 3 operations
        # If min_char or max_char exists in the internal substring s[1:n-1]
        if min_char in s[1 : n - 1] or max_char in s[1 : n - 1]:
            return 2

        return 3
```
