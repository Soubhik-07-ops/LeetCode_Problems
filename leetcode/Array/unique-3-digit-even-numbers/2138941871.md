```
class Solution:
    def totalNumbers(self, digits: List[int]) -> int:
        freq = Counter(digits)
        count = 0
        
        for num in range(100, 1000, 2):  # all 3-digit even numbers
            num_freq = Counter(str(num))
            # Check if we have enough of each digit
            if all(freq[int(d)] >= cnt for d, cnt in num_freq.items()):
                count += 1
        
        return count
```
