```
from typing import List

class Solution:
    def missingMultiple(self, nums: List[int], k: int) -> int:
        # Convert the input array to a set for efficient lookups
        nums_set = set(nums)
        
        # Find the first missing multiple
        multiple = k
        while True:
            if multiple not in nums_set:
                return multiple
            multiple += k
```
