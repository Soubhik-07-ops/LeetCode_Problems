class Solution {
    public boolean predictTheWinner(int[] nums) {
        int n = nums.length;
        Integer[][] memo = new Integer[n][n];
        return dfs(nums, 0, n - 1, memo) >= 0;
    }
    
    private int dfs(int[] nums, int i, int j, Integer[][] memo) {
        if (i > j) return 0;
        if (memo[i][j] != null) return memo[i][j];
        
        // If we pick nums[i], we get nums[i] minus what opponent can get from remaining
        int pickLeft = nums[i] - dfs(nums, i + 1, j, memo);
        int pickRight = nums[j] - dfs(nums, i, j - 1, memo);
        
        memo[i][j] = Math.max(pickLeft, pickRight);
        return memo[i][j];
    }
}
