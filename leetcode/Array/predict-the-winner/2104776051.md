class Solution {
    fun predictTheWinner(nums: IntArray): Boolean {
        val n = nums.size
        val memo = Array(n) { IntArray(n) { Int.MIN_VALUE } }
        
        // Returns the maximum score difference (current player - opponent)
        // that the current player can achieve from subarray nums[i..j]
        fun dfs(i: Int, j: Int): Int {
            if (i > j) return 0
            if (memo[i][j] != Int.MIN_VALUE) return memo[i][j]
            
            // If we pick nums[i], we get nums[i] plus the negative of what opponent can get from remaining array
            val pickLeft = nums[i] - dfs(i + 1, j)
            val pickRight = nums[j] - dfs(i, j - 1)
            
            memo[i][j] = maxOf(pickLeft, pickRight)
            return memo[i][j]
        }
        
        // Player 1 wins if the score difference is >= 0
        return dfs(0, n - 1) >= 0
    }
}
