class Solution:
    def resultArray(self, nums: List[int], k: int) -> List[int]:
        result = [0] * k
        n = len(nums)
        
        # prev_counts[r] = number of subarrays ending at previous position with remainder r
        prev_counts = [0] * k
        
        for i, num in enumerate(nums):
            curr_counts = [0] * k
            rem = num % k
            
            # Subarray consisting of just nums[i]
            curr_counts[rem] += 1
            
            # Extend all subarrays ending at i-1 by nums[i]
            for r in range(k):
                if prev_counts[r] > 0:
                    new_rem = (r * rem) % k
                    curr_counts[new_rem] += prev_counts[r]
            
            # Add current counts to result
            for r in range(k):
                result[r] += curr_counts[r]
            
            prev_counts = curr_counts
        
        return result
