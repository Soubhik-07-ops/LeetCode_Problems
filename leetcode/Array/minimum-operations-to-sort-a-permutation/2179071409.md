class Solution:

  def minOperations(self, nums: list[int]) -> int:
    n = len(nums)
    if n <= 1:
      return 0

    k = nums.index(0)

    # Check if nums is an increasing cyclic shift: [0, 1, ..., n - 1]
    is_inc = all(nums[(k + i) % n] == i for i in range(n))
    if is_inc:
      # Option 1: Rotate left k times
      # Option 2: Reverse, rotate left (n - k) % n times, reverse
      return min(k, 2 + (n - k) % n)

    # Check if nums is a decreasing cyclic shift: [0, n - 1, ..., 1]
    is_dec = all(nums[(k - i + n) % n] == i for i in range(n))
    if is_dec:
      # Option 1: Reverse then rotate left (n - 1 - k) times
      # Option 2: Rotate left (k + 1) % n times then reverse
      return min(1 + (n - 1 - k), 1 + (k + 1) % n)

    return -1
