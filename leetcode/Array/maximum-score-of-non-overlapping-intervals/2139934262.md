from bisect import bisect_right
from functools import lru_cache
from typing import List

class Solution:
    def maximumWeight(self, intervals: List[List[int]]) -> List[int]:
        # Store (l, r, weight, original_index)
        n = len(intervals)
        arr = [(intervals[i][0], intervals[i][1], intervals[i][2], i) for i in range(n)]
        arr.sort()
        
        starts = [x[0] for x in arr]
        
        @lru_cache(maxsize=None)
        def dp(i: int, k: int):
            if i == n or k == 0:
                return (0, ())
            
            # Option 1: Skip interval i
            best_weight, best_indices = dp(i + 1, k)
            
            # Option 2: Take interval i
            l, r, weight, orig_idx = arr[i]
            next_i = bisect_right(starts, r)
            next_weight, next_indices = dp(next_i, k - 1)
            
            take_weight = weight + next_weight
            # Result indices should be in sorted order
            take_indices = tuple(sorted((orig_idx, *next_indices)))
            
            # Compare taking vs skipping:
            # 1. Higher total weight wins
            # 2. On tie, lexicographically smaller index list wins
            if take_weight > best_weight:
                best_weight, best_indices = take_weight, take_indices
            elif take_weight == best_weight and take_indices < best_indices:
                best_weight, best_indices = take_weight, take_indices
                
            return (best_weight, best_indices)
        
        return list(dp(0, 4)[1])
