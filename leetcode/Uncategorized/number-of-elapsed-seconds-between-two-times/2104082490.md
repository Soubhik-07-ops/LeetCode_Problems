class Solution:
    def secondsBetweenTimes(self, startTime: str, endTime: str) -> int:
        # Parse start time
        start_hours = int(startTime[0:2])
        start_minutes = int(startTime[3:5])
        start_seconds = int(startTime[6:8])
        
        # Parse end time
        end_hours = int(endTime[0:2])
        end_minutes = int(endTime[3:5])
        end_seconds = int(endTime[6:8])
        
        # Convert to total seconds from midnight
        start_total = start_hours * 3600 + start_minutes * 60 + start_seconds
        end_total = end_hours * 3600 + end_minutes * 60 + end_seconds
        
        # Return the difference
        return end_total - start_total
