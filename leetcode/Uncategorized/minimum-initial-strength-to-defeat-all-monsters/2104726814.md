```
class Solution:
    def minInitialStrength(self, monsters: list[int], boosts: list[list[int]]) -> int:
        n = len(monsters)
        
        # Compute bonus for each position using difference array
        diff = [0] * (n + 1)
        for l, r, v in boosts:
            diff[l] += v
            diff[r + 1] -= v
        
        bonus = [0] * n
        current = 0
        for i in range(n):
            current += diff[i]
            bonus[i] = current
        
        # Work backwards to find minimum initial strength
        needed = 0
        for i in range(n - 1, -1, -1):
            if needed == 0:
                needed = max(0, monsters[i] - bonus[i])
            else:
                needed = monsters[i] + needed
        
        return needed
```
