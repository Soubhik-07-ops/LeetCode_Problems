```
class Solution:
    def weightedSum(self, parent: list[int], nums: list[int]) -> int:
        n = len(parent)
        
        # Build adjacency list for the tree
        children = [[] for _ in range(n)]
        for i in range(1, n):
            children[parent[i]].append(i)
        
        # Find height and depths using DFS
        depths = [0] * n
        height = 0
        
        def dfs(node, depth):
            nonlocal height
            depths[node] = depth
            height = max(height, depth)
            for child in children[node]:
                dfs(child, depth + 1)
        
        dfs(0, 1)  # root has depth 1
        
        # Calculate sum of weights
        total = 0
        for i in range(n):
            total += nums[i] * (height - depths[i] + 1)
        
        return total
```
