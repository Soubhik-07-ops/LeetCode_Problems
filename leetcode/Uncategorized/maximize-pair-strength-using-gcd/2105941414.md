```
from typing import List
from math import gcd

class Solution:
    def maxPairStrength(self, nums: List[int]) -> int:
        max_strength = 0
        n = len(nums)
        
        # Brute force check all pairs
        for i in range(n):
            for j in range(i + 1, n):
                g = gcd(nums[i], nums[j])
                strength = (nums[i] * nums[j]) // (g * g)
                max_strength = max(max_strength, strength)
        
        return max_strength
```
